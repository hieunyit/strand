package web
import("testing";"strings";"context";"os";"time")
func TestAuditUnmaskReordered(t *testing.T){
 orig:=[]byte("source:\n  password: source-secret\ntarget:\n  password: target-secret\n")
 edited:="target:\n  password: \"********\"\nsource:\n  password: \"********\"\n"
 got,err:=UnmaskConfigText(edited,orig);if err!=nil{t.Fatal(err)}
 if !strings.HasPrefix(got,"target:\n  password: target-secret"){t.Errorf("passwords assigned by occurrence rather than YAML path: %s",got)}
}
func TestAuditEnvSavePersistsRegistration(t *testing.T){
 e:=newEnv(t,nil);p:=e.srv.reg.Get("orders")
 if _,err:=e.srv.saveEnvContent(context.Background(),p,map[string]string{"SRC_PW":"test-secret"},"test");err!=nil{t.Fatal(err)}
 if e.srv.reg.Get("orders").EnvFile==""{t.Error("saveEnvContent returns success but Refresh loses new env_file")}
}
func TestAuditSecretsInAudit(t *testing.T){
 e:=opsEnv(t,false,nil);c,csrf:=e.loginOK()
 code,_:=e.act(c,csrf,"orders","env-save",map[string]string{"entries":`{"SRC_PW":"audit-canary-secret"}`});if code!=200{t.Fatalf("status %d",code)}
 raw,err:=os.ReadFile(e.srv.audit.Path());if err!=nil{t.Fatal(err)}
 if strings.Contains(string(raw),"audit-canary-secret"){t.Error("plaintext password saved into audit log")}
}
func TestAuditConfigDiffLeaksSecret(t *testing.T){
 e:=opsEnv(t,false,nil);p:=e.srv.reg.Get("orders")
 raw,err:=os.ReadFile(p.ConfigPath);if err!=nil{t.Fatal(err)}
 raw=append(raw,[]byte("\n# comment\npassword: diff-canary-secret\n")...)
 if err=os.WriteFile(p.ConfigPath,raw,0600);err!=nil{t.Fatal(err)}
 masked,err:=MaskConfigText(raw);if err!=nil{t.Fatal(err)}
 res,err:=e.srv.validateConfigContent(context.Background(),p,masked+"# next\n");if err!=nil{t.Fatal(err)}
 if strings.Contains(res.Diff,"diff-canary-secret"){t.Error("config diff discloses originally masked password")}
}
func TestAuditLimiterMemory(t *testing.T){
 now:=time.Now();l:=NewLoginLimiter(func()time.Time{return now});l.Fail("192.0.2.1");now=now.Add(24*time.Hour);l.Fail("192.0.2.2")
 if _,ok:=l.m["192.0.2.1"];ok{t.Error("expired failed IP is never removed")}
}
