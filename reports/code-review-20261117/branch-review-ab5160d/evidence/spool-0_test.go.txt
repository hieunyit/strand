package spool
import (
 "context"
 "os"
 "path/filepath"
 "testing"
)
func TestSurfaceStaleInventoryOverwritesLiveSize(t *testing.T) {
 temp:=t.TempDir(); trusted:=filepath.Join(temp,"trusted"); dbdir:=filepath.Join(temp,"db")
 if err:=os.Mkdir(trusted,0700);err!=nil {t.Fatal(err)}
 path:=writeNamedFile(t,trusted,"orphan",100)
 s,err:=New(dbdir);if err!=nil {t.Fatal(err)}
 report,err:=s.ScanInventory([]string{trusted});if err!=nil {t.Fatal(err)}
 if err:=os.WriteFile(path,make([]byte,200),0600);err!=nil {t.Fatal(err)}
 if err:=s.Append(context.Background(),spillTxn(path,"1"));err!=nil {t.Fatal(err)}
 if _,err:=s.ImportInventory(report);err!=nil {t.Fatal(err)}
 t.Logf("live sidecar: indexed bytes=%d actual bytes=200 refs=%d",s.sidecars[path].size,s.sidecars[path].refs)
 if err:=s.Close();err!=nil {t.Fatal(err)}
 reopened,err:=New(dbdir)
 if err!=nil {t.Fatalf("legal scan->grow->append->import sequence prevented spool reopen: %v",err)}
 reopened.Close()
}
func TestSurfacePostVersionCrashOrphanNotCharged(t *testing.T) {
 dbdir:=filepath.Join(t.TempDir(),"db")
 s,err:=New(dbdir);if err!=nil {t.Fatal(err)}
 if err:=s.Close();err!=nil {t.Fatal(err)}
 spilldir:=filepath.Join(dbdir,"spill")
 if err:=os.Mkdir(spilldir,0700);err!=nil {t.Fatal(err)}
 // A completed or partial assembler spill surviving a crash before Append persists its path.
 writeNamedFile(t,spilldir,"strand-crashed.spill",4096)
 s,err=New(dbdir,WithTotalHardLimit(1024));if err!=nil {t.Fatal(err)}
 defer s.Close()
 t.Logf("physical retained spill=4096B, reported local=%d",s.Usage().LocalBytes)
 if s.Usage().LocalBytes<4096 {t.Error("known default-directory crash orphan was not indexed on restart")}
 tx:=spillTxn("","2")
 if err:=s.Append(context.Background(),tx);err==nil {t.Error("append allowed despite retained sidecar already exceeding total limit")}
}
